\begin{exercise}
    Suppose $A$ is an $m \times n$ matrix, $m >n$.  
    \begin{enumerate}
        \item What is the computational cost of computing $A^k$ for some $k \in \mathbb{N}$?
        \item Now, write $A$ using its SVD.  How can you simplify $A^k$ by using the SVD, and using this transformation, what is the computational cost of computing $A^k$? (Hint: the computational cost of computing the SVD is $\mathcal{O}(mn^2)$.)
    \end{enumerate}
\end{exercise}

\begin{exercise}
    What is the rank of $A = U\Sigma V^*$?
\end{exercise}

\begin{exercise}
    Compute the determinant of $A = U\Sigma V^*$.
\end{exercise}
