\section{Floating Point Operation Problems}
Assume IEEE arithmetic with round-to-nearest, ties-to-even, and that no overflow or underflow occurs, so that
\begin{align*}
    \text{fl}(x \; \odot\; y) = (x \; \odot\; y)(1+\delta)
\end{align*}
where $|\delta| < \epsilon$, $\epsilon$ represents the maximum relative representation error and $\odot = +, - , \times, /$.
\begin{exercise}
    Show whether or not the following statements are true in IEEE arithmetic, assuming $a$ and $b$ are normalized floating point numbers (and that no exception occurs in the stated operations)?
    \begin{enumerate}
        \item $\text{fl}(a \;\odot\; b) = \text{fl}(b \;\odot\; a)$, $\odot = +, \times$
        \item $\text{fl}(b - a) = - \text{fl}(a-b)$
        \item $\text{fl}(a + a) = \text{fl}(2a)$
        \item $\text{fl}(0.5 \times a) = \text{fl}(a/2)$
        \item $\text{fl}((a+b)+c) = \text{fl}(a+(b+c))$
        \item $a \leq \text{fl}((a+b)/2) \leq b$ if $a \leq b$
    \end{enumerate}
\end{exercise}

\begin{exercise}
    Let $L$ be a lower triangular matrix and solve $Lx = b$ by forward substitution. Show that, barring overflow or underflow, the computed solution $\hat{x}$ satisfies $(L+\delta L) \hat{x} = b$ (where the elements of $\delta L$ are denoted $\delta l_{ij}$ and the elements of $L$ are denoted $l_{ij}$), where $|\delta l_{ij}| \leq n\epsilon |l_{ij}|$, where $\epsilon$ is the machine precision (where in the upper bound we are neglecting all additional terms of order $\epsilon^p$, $p \geq 2$) . This means that forward substitution is backward stable. Argue that backward substitution for solving upper triangular systems satisfies the same bound.
\end{exercise}

\section{Vector and Matrix Norm Problems}
\begin{exercise}
    Verify that $\|xy^*\|_F = \|xy^*\|_2 = \|x\|_2\|y\|_2$ for any $x, y \in \mathbb{C}^n$.
\end{exercise}

\begin{exercise}
    \begin{itemize} 
    \item Prove that $\|Ax\| \leq \|A\|\cdot \|x\|$ for a vector norm and its corresponding operator norm.
    \item Prove that $\|Ax\|_2 \leq \|A\|_F\|x\|_2$.
    \item Prove that $\|AB\| \leq \|A\|\|B\|$.
    \item If $A \in \mathbb{C}^{n\times n}$, then $n^{-1/2}\|A\|_2 \leq \|A\|_1 \leq n^{1/2}\|A\|_2$.
    \end{itemize}
\end{exercise}

\begin{exercise}
    Suppose $Ax = b$ and  $(A+\delta A)\hat{x} = b+ \delta b$.  Define $\delta x := \hat{x} -x$.  Show that 
    \begin{align*}
        \|\delta x\| \leq \|A^{-1}\|(\|\delta A\| \cdot \|\hat{x}\| + \|\delta b\|)
    \end{align*}
\end{exercise}

\section{SVD, Projections, Gram-Schmidt, and QR}

\begin{exercise}
    Circle the correct answer.
    \begin{enumerate}
    \item What is the purpose of computing $A^T A$?

    \begin{enumerate}[label=(\alph*)]
        \item To find the eigenvectors of $A$ directly.
        \item To obtain a symmetric matrix whose eigenvalues
        give the squared singular values.
        \item To make $A$ a square matrix.
    \end{enumerate}

    \vspace{0.5cm}

    \item If an eigenvalue of $A^T A$ is $9$, what is the
    corresponding singular value?

    \begin{enumerate}[label=(\alph*)]
        \item $3$
        \item $9$
        \item $81$
    \end{enumerate}

    \item Why must zero singular values be handled carefully
    when computing $U$ using
    $U = AV\Sigma^{-1}$?

    \begin{enumerate}[label=(\alph*)]
        \item Because the inverse of $\Sigma$ may not exist.
        \item Because $V$ must be square.
        \item Because $A$ must be symmetric.
    \end{enumerate}

    \end{enumerate}
\end{exercise}

\begin{exercise}

\[
A = QR,
\]
where $Q \in \mathbb{R}^{m \times n}$ has orthonormal columns
and $R \in \mathbb{R}^{n \times n}$ is upper triangular.

\begin{enumerate}
    \item Fill in each blank to complete the Modified Gram--Schmidt

and \texttt{norm(x,2)} computes the Euclidean norm.

\begin{lstlisting}

    Input: A in R^(m x n), with linearly independent columns
    Output: Q in R^(m x n), R in R^(n x n)

    V <- __________ (1) __________

    R <- zero matrix of size n x n

    for k = 1, 2, ..., n:

        R[k,k] <- __________ (2) __________
        Q[:,k] <- __________ (3) __________

        for j = k+1, k+2, ..., n:

            V[:,j] <- __________ (5) __________

\item Answer the following questions.

    \item Why is $R_{kk}$ computed using a vector norm?
    \item What is the purpose of updating $V_{:,j}$ inside

    \item Why is $R$ upper triangular?
\end{enumerate}

\item Give the leading-order arithmetic complexity of the algorithm
in terms of $m$ and $n$, assuming $m \geq n$.

\end{enumerate}
