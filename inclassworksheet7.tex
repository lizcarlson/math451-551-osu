    \begin{abstract} Solve the following problems.
    \end{abstract}
    \maketitle

\begin{exercise}
    Consider a general hermitian positive definite $2\times 2$ matrix with first entry $1$:
    \begin{align*}
        A = \begin{bmatrix}
            1 & w_1^* \\
            w_1 & K_1
        \end{bmatrix}.
    \end{align*}
    Find the elimination matrix $L_1$, and compute $L_1A$.  Then, write the LU decomposition of $A$, where $L = L_1^{-1}$.
    \vfill
\end{exercise}

\begin{exercise}
    Consider $U^*$, which will be of the form
    \begin{align*}
        U^* = \begin{bmatrix}
            * & 0 \\ * & *
        \end{bmatrix}.
    \end{align*}
    Compute $L_1 U^*$.  What is the resulting structure of this matrix?
    \vfill
\end{exercise}
\newpage

\begin{exercise}
    Define $D:= L_1U^*$, and note that $D^* = UL_1^*$.  Notice that 
    $A = L_1^{-1}U$.  Write $A$ in terms of $L_1, L_1^{-1}$, and/or $D$.
\end{exercise}

\vfill

\begin{exercise}
    Using \#3, write out $A$ as a product of three matrices: a lower triangular matrix, a diagonal matrix, and an upper triangular matrix.
\end{exercise}
\vfill
\newpage
\begin{exercise}
    Redo \#1 with now $w_1$ is a $2\times 1$ vector instead of a scalar, and $K_1$ is a $2 \times 2$ hermitian positive definite matrix.  Does anything change in the structure of your analysis? 
\end{exercise}

\vfill

\begin{exercise} (\textbf{Bonus})
Finish exercises \#1-\#4 with $w_1$ being a general $n \times 1$ vector.    
\end{exercise}
